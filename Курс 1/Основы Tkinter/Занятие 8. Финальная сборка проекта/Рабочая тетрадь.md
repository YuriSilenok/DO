### Шаг 1. Приводим код в порядок: полный листинг

**Что делаем.** Создаём новый чистый `main.py` и собираем в нём всё, что изучили за 8 занятий:

```python
import io
import os
import tempfile
import threading
import time
import tkinter as tk
from tkinter import filedialog, messagebox
import requests
from PIL import Image, ImageTk

BASE_URL = 'http://127.0.0.1:8000'
HEALTH_URL = BASE_URL + '/health'


class ChatApp:
    """Анонимный чат на Tkinter + chat_api."""

    def __init__(self, root):
        self.root = root
        root.title('Анонимный чат')
        root.geometry('460x640')

        self.username = ''
        self.chat_id = 0
        self.password = ''
        self.last_message_id = 0
        self._images = []

        self.auto_scroll = tk.BooleanVar(value=True)
        self.notify_mode = tk.StringVar(value='все')

        self._build_widgets()

    def _build_widgets(self):
        self.welcome = tk.Label(self.root, text='Добро пожаловать в чат!', font=('Arial', 14))
        self.welcome.pack(pady=10)

        self.info = tk.Label(self.root, text='Введите номер чата, имя и пароль')
        self.info.pack()

        self.chat_entry = tk.Entry(self.root, width=20)
        self.chat_entry.pack(pady=2)

        self.name_entry = tk.Entry(self.root, width=20)
        self.name_entry.pack(pady=2)

        self.pass_entry = tk.Entry(self.root, width=20, show='*')
        self.pass_entry.pack(pady=2)

        self.enter_btn = tk.Button(self.root, text='Войти в чат', command=self.on_enter)
        self.enter_btn.pack(pady=10)

        self.check_btn = tk.Button(self.root, text='Проверить сервер', command=self.check_server)
        self.check_btn.pack(pady=5)

        self.chat_view = tk.Text(self.root, height=8, state='disabled')
        self.chat_view.pack(pady=5, padx=5, fill='both', expand=True)

        self.auto_check = tk.Checkbutton(
            self.root, text='Автопрокрутка ленты',
            variable=self.auto_scroll,
        )
        self.auto_check.pack(pady=2)

        self.radio_frame = tk.Frame(self.root)
        self.radio_frame.pack(pady=2)

        tk.Radiobutton(
            self.radio_frame, text='Все сообщения', value='все',
            variable=self.notify_mode,
        ).pack(side='left', padx=4)

        tk.Radiobutton(
            self.radio_frame, text='Только текст', value='только текст',
            variable=self.notify_mode,
        ).pack(side='left', padx=4)

        self.msg_entry = tk.Entry(self.root, width=30)
        self.msg_entry.pack(pady=2)

        self.send_btn = tk.Button(self.root, text='Отправить', command=self.send_message)
        self.send_btn.pack(pady=5)

        self.file_btn = tk.Button(self.root, text='Отправить картинку', command=self.send_file_message)
        self.file_btn.pack(pady=5)

        self.info_btn = tk.Button(self.root, text='О себе', command=self.show_about)
        self.info_btn.pack(pady=5)

        self.file_frame = tk.Frame(self.root, bd=1, relief='sunken')
        self.file_frame.pack(fill='x', padx=5, pady=5)
        tk.Label(self.file_frame, text='Загруженные файлы:').pack()
        self.file_list = tk.Listbox(self.file_frame, height=3)
        self.file_list.pack(fill='x', padx=5, pady=2)
        self.file_list.insert(1, 'Пока ничего не загружено')
        self.clear_list_btn = tk.Button(
            self.file_frame, text='Удалить выбранный файл из списка',
            command=self.clear_file_list,
        )
        self.clear_list_btn.pack(pady=2)

    def set_status(self, text):
        self.info.config(text=text)

    def check_server(self):
        """Проверка сервера: GET /health."""
        try:
            response = requests.get(HEALTH_URL, timeout=5)
            if response.status_code == 200:
                self.set_status('Сервер чата работает: {0}'.format(response.json()))
            else:
                self.set_status('Сервер ответил кодом {0}'.format(response.status_code))
        except requests.exceptions.RequestException:
            self.set_status('Ошибка сети: запустите сервер chat_api (python -m chat_api)')

    def on_enter(self):
        """Вход в чат: POST /chats/join + запуск опроса."""
        chat_text = self.chat_entry.get().strip()
        name = self.name_entry.get().strip()
        password = self.pass_entry.get()
        if not chat_text or not name or not password:
            self.set_status('Заполните все поля: чат, имя, пароль!')
            return

        try:
            self.set_status('Подключаюсь к чату {0}...'.format(chat_text))
            response = requests.post(
                BASE_URL + '/chats/join',
                json={
                    'chat_id': int(chat_text),
                    'username': name,
                    'password': password,
                },
                timeout=10,
            )
            data = response.json()
            if response.status_code == 200:
                self.username = data['username']
                self.chat_id = data['chat_id']
                self.password = password
                self.set_status(
                    'Вы вошли как {0} в чат №{1}'.format(self.username, self.chat_id)
                )
                self.start_polling()
            else:
                self.set_status('Ошибка {0}: {1}'.format(
                    response.status_code, data.get('detail', 'неизвестная ошибка')
                ))
        except ValueError:
            self.set_status('Номер чата должен быть целым числом!')
        except requests.exceptions.RequestException:
            self.set_status('Ошибка сети: запустите сервер chat_api (python -m chat_api)')

    def append_message(self, text):
        self.chat_view.config(state='normal')
        self.chat_view.insert(tk.END, text + '\n')
        if self.auto_scroll.get():
            self.chat_view.see(tk.END)
        self.chat_view.config(state='disabled')

    def send_message(self):
        """Отправка текста: POST /chats/send."""
        text = self.msg_entry.get().strip()
        if not text:
            self.set_status('Введите текст сообщения')
            return
        if not self.chat_id:
            self.set_status('Сначала войдите в чат!')
            return
        try:
            response = requests.post(
                BASE_URL + '/chats/send',
                json={
                    'chat_id': self.chat_id,
                    'username': self.username,
                    'password': self.password,
                    'text': text,
                },
                timeout=10,
            )
            data = response.json()
            if response.status_code == 200:
                message = data['message']
                self.append_message('{0}: {1}'.format(self.username, message['text']))
                self.msg_entry.delete(0, tk.END)
                self.set_status('Сообщение отправлено (id {0})'.format(message['id']))
            else:
                self.set_status('Ошибка {0}: {1}'.format(
                    response.status_code, data.get('detail', 'неизвестная ошибка')
                ))
        except requests.exceptions.RequestException:
            self.set_status('Ошибка сети: запустите сервер chat_api (python -m chat_api)')

    def send_text(self, text):
        self.msg_entry.delete(0, tk.END)
        self.msg_entry.insert(0, text)
        self.send_message()

    def send_file_message(self):
        """Выбор файла и отправка: POST /chats/send_file."""
        path = filedialog.askopenfilename(
            title='Выберите картинку',
            filetypes=[('Картинки', '*.png *.jpg *.jpeg *.gif'), ('Все файлы', '*.*')],
        )
        if not path:
            self.set_status('Выбор файла отменён')
            return
        self.upload_image(path)

    def upload_image(self, path):
        """Отправка картинки: POST /chats/send_file."""
        if not self.chat_id:
            self.set_status('Сначала войдите в чат!')
            return
        try:
            files = {'file': open(path, 'rb')}
            data = {
                'chat_id': self.chat_id,
                'username': self.username,
                'password': self.password,
            }
            response = requests.post(
                BASE_URL + '/chats/send_file',
                data=data,
                files=files,
                timeout=30,
            )
            response.raise_for_status()
            file_id = response.json()['message']['id']
            self.file_list.insert(tk.END, 'файл #{0} — {1}'.format(file_id, os.path.basename(path)))
            self.file_list.select_clear(0, tk.END)
            self.file_list.select_set(tk.END)
            self.set_status('Картинка отправлена (id {0})'.format(file_id))
        except requests.exceptions.RequestException:
            self.set_status('Ошибка сети или неверный формат картинки')

    def clear_file_list(self):
        """Удаление выбранного файла из списка Listbox."""
        selection = self.file_list.curselection()
        if selection:
            self.file_list.delete(selection[0])
            self.set_status('Строка удалена из списка')
        else:
            self.set_status('Ничего не выбрано в списке')

    def add_image_to_view(self, path):
        """Вставка уменьшенной картинки в ленту Text."""
        try:
            image = Image.open(path)
            image.thumbnail((150, 150))
            photo = ImageTk.PhotoImage(image)

            self.chat_view.config(state='normal')
            self.chat_view.image_create(tk.END, image=photo)
            self.chat_view.insert(tk.END, '\n')
            if self.auto_scroll.get():
                self.chat_view.see(tk.END)
            self.chat_view.config(state='disabled')

            self._images.append(photo)
        except Exception:
            self.chat_view.config(state='normal')
            self.chat_view.insert(tk.END, '[Не удалось показать картинку]\n')
            self.chat_view.config(state='disabled')

    def load_image_by_url(self, url):
        """Скачивание картинки с сервера: GET /files/{id}."""
        try:
            response = requests.get(BASE_URL + url, timeout=10)
            if response.status_code == 200:
                image = Image.open(io.BytesIO(response.content))
                tmp = os.path.join(
                    tempfile.gettempdir(),
                    'msg_{0}.tmp'.format(len(self._images)),
                )
                image.save(tmp)
                self.add_image_to_view(tmp)
                return True
        except Exception:
            pass
        return False

    def poll_once(self):
        """Получение новых сообщений: POST /chats/poll."""
        try:
            response = requests.post(
                BASE_URL + '/chats/poll',
                json={
                    'chat_id': self.chat_id,
                    'username': self.username,
                    'password': self.password,
                    'last_message_id': self.last_message_id,
                    'timeout': 2,
                },
                timeout=10,
            )
            data = response.json()
            if response.status_code == 200:
                for m in data['messages']:
                    if self.notify_mode.get() == 'только текст' and m.get('file_url'):
                        self.last_message_id = m['id']
                        continue
                    self.append_message(
                        '{0}: {1}'.format(m['username'], m['text'] or '')
                    )
                    if m.get('file_url'):
                        self.append_message('{0} прислал(а) картинку:'.format(m['username']))
                        self.load_image_by_url(m['file_url'])
                    self.last_message_id = m['id']
                return True
        except requests.exceptions.RequestException:
            pass
        return False

    def start_polling(self):
        """Фоновый поток опроса новых сообщений."""
        def worker():
            while True:
                if not self.poll_once() and not self.poll_once():
                    time.sleep(1)
                time.sleep(1)

    def show_about(self):
        """Окно «О себе» с круглой аватаркой и GET /files/{id}."""
        if not self.chat_id:
            self.set_status('Сначала войдите в чат!')
            return
        win = tk.Toplevel(self.root)
        win.title('О себе')
        win.geometry('320x250')

        tk.Label(win, text='Вы вошли как {0}'.format(self.username)).pack(pady=10)

        canvas = tk.Canvas(win, width=120, height=120, bg='white', highlightthickness=0)
        canvas.pack(pady=10)

        self.download_avatar(canvas, 60)

        tk.Button(win, text='Закрыть', command=win.destroy).pack(pady=10)

    def download_avatar(self, canvas, radius):
        """Скачивание и обрезка аватарки в круг: GET /files/{id}."""
        file_url = BASE_URL + '/files/1'
        try:
            response = requests.get(file_url, timeout=10)
            if response.status_code != 200:
                self.draw_letter_avatar(canvas, radius)
                return

            image = Image.open(io.BytesIO(response.content))
            image = image.resize((radius * 2, radius * 2))
            self._images.append(ImageTk.PhotoImage(image))

            canvas.create_image(radius, radius, image=self._images[-1])
            canvas.create_oval(0, 0, radius * 2, radius * 2, outline='#333333', width=2)
        except Exception:
            self.draw_letter_avatar(canvas, radius)

    def draw_letter_avatar(self, canvas, radius):
        """Аватарка с буквой инициала, если файла /files/1 нет."""
        canvas.delete('all')
        canvas.create_oval(0, 0, radius * 2, radius * 2, fill='#5b9bd5', outline='')
        initial = (self.username[:1] or '?').upper()
        canvas.create_text(
            radius, radius, text=initial, fill='white',
            font=('Arial', radius, 'bold'),
        )


app = ChatApp(tk.Tk())
app.root.mainloop()
```

### Шаг 3. Защита от ошибок ввода

**Что делаем.** Ещё раз проверяем, что программа не «падает» на вводе пользователя. Обойдите все места ввода и убедитесь, что:

| Ситуация | Как защищаемся | Где в коде |
|---|---|---|
| Пустые поля при входе | проверка `if not ...` | `on_enter` |
| Номер чата не число | `int(chat_text)` в `try/except ValueError` | `on_enter` |
| Сервер не запущен | `except requests.exceptions.RequestException` | все сетевые методы |
| Не выбран файл | проверка `if not path:` | `send_file_message` |
| «Битый» файл-картинка | `try/except` в `add_image_to_view` | `add_image_to_view` |
| Пустое сообщение | `if not text:` | `send_message` |
| Выход без входа | `if not self.chat_id:` | `send_message`, `send_file_message`, `show_about` |

**Что здесь происходит.** Мы не доверяем вводу пользователя: **валидация** (проверка пустых полей, типов) и **обработка исключений** (сеть, картинки) превращают вероятные ошибки в аккуратные подсказки в строке статуса вместо «краша» программы с красной ошибкой.

**Проверь себя:** кликните «Войти в чат» с пустыми полями; введите буквы в номер чата; запустите программу без сервера — ни в одном случае программа не падает.
