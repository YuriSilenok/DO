## Шаг 1. Поле ввода и список
```python
import tkinter as tk

root = tk.Tk()
root.title('Список дел')
root.geometry('350x300')

entry = tk.Entry(root, width=30)
entry.pack()

listbox = tk.Listbox(root, width=30, height=10)
listbox.pack()

def add_task():
    task = entry.get()
    if task:
        listbox.insert(tk.END, task)
        entry.delete(0, tk.END)

button = tk.Button(root, text='Добавить', command=add_task)
button.pack()

root.mainloop()
```

**Пояснения к новым строкам:**
- `def add_task():` — объявляет функцию добавления задачи.
- `if task:` — проверяет, что поле не пустое.
- `entry.delete(0, tk.END)` — очищает поле ввода от первого символа до конца.

## Шаг 2. Добавляем удаление выбранного пункта
```python
def delete_task():
    selected = listbox.curselection()
    if selected:
        listbox.delete(selected[0])
```

**Пояснения к новым строкам:**
- `def delete_task():` — объявляет функцию для удаления выбранной задачи.
- `selected = listbox.curselection()` — получает индекс выбранного пункта; если ничего не выбрано, результат будет пустым.
- `if selected:` — проверяет, что пользователь действительно выбрал пункт.

## Шаг 3. Превращаем список дел в чат-клиент

Теперь вместо задач пользователь добавляет **сообщения**, а список —
это **переписка**. Получение новых сообщений выполняется в отдельном
потоке, чтобы интерфейс не зависал. Используем только стандартную
библиотеку: `tkinter`, `json`, `urllib.request`, `threading`.

```python
import json
import threading
import tkinter as tk
import urllib.request
import urllib.error

SERVER = 'http://127.0.0.1:8000'

root = tk.Tk()
root.title('Чат-клиент')
root.geometry('460x420')

# --- Поля входа -------------------------------------------------------
label_room = tk.Label(root, text='Номер чата:')
label_room.pack()
entry_room = tk.Entry(root, width=30)
entry_room.pack()

label_name = tk.Label(root, text='Имя:')
label_name.pack()
entry_name = tk.Entry(root, width=30)
entry_name.pack()

label_password = tk.Label(root, text='Пароль:')
label_password.pack()
entry_password = tk.Entry(root, width=30, show='*')
entry_password.pack()

# --- Поле ввода сообщения ---------------------------------------------
entry_text = tk.Entry(root, width=30)
entry_text.pack()

# --- Окно переписки ---------------------------------------------------
chat_log = tk.Text(root, width=55, height=12, state='disabled')
chat_log.pack()

status = tk.Label(root, text='Ещё не вошли в чат')
status.pack()

last_message_id = 0
joined = False


def post(path, payload):
    """Отправляет POST-запрос на сервер и возвращает разобранный JSON."""
    body = json.dumps(payload).encode('utf-8')
    request = urllib.request.Request(
        SERVER + path,
        data=body,
        headers={'Content-Type': 'application/json'},
        method='POST',
    )
    with urllib.request.urlopen(request, timeout=10) as response:
        return json.load(response)


def append_message(message):
    """Добавляет одно сообщение в лог чата."""
    chat_log.config(state='normal')
    chat_log.insert(
        tk.END,
        '{}: {}\n'.format(message['username'], message.get('text') or '(вложение)'),
    )
    chat_log.config(state='disabled')


def enter_chat():
    global joined
    try:
        room = int(entry_room.get())
        name = entry_name.get()
        password = entry_password.get()

        post('/chats/join', {
            'chat_id': room,
            'username': name,
            'password': password,
        })
        joined = True
        status.config(text='Вы вошли в чат ' + str(room))
        # начинаем следить за новыми сообщениями
        threading.Thread(target=poll_loop, daemon=True).start()
    except Exception as exc:
        status.config(text='Ошибка входа: ' + str(exc))


def send_message():
    try:
        room = int(entry_room.get())
        name = entry_name.get()
        password = entry_password.get()
        text = entry_text.get()

        post('/chats/send', {
            'chat_id': room,
            'username': name,
            'password': password,
            'text': text,
        })
        entry_text.delete(0, tk.END)
    except Exception as exc:
        status.config(text='Ошибка отправки: ' + str(exc))


def poll_loop():
    """Фоновый поток: ждёт новые сообщения и показывает их в окне."""
    global last_message_id
    while joined:
        try:
            room = int(entry_room.get())
            name = entry_name.get()
            password = entry_password.get()

            answer = post('/chats/poll', {
                'chat_id': room,
                'username': name,
                'password': password,
                'last_message_id': last_message_id,
                'timeout': 2,
            })
            for message in answer['messages']:
                root.after(0, append_message, message)
                last_message_id = max(last_message_id, message['id'])
        except Exception:
            pass


button_enter = tk.Button(root, text='Войти в чат', command=enter_chat)
button_enter.pack()

button_send = tk.Button(root, text='Отправить', command=send_message)
button_send.pack()

root.mainloop()
```

**Пояснения к новым строкам:**
- `import threading` — подключает модуль для запуска фоновых потоков.
- `def post(path, payload):` — вспомогательная функция: отправляет запрос и возвращает ответ в виде словаря. Мы написали её один раз, чтобы не повторять код в каждой функции.
- `chat_log = tk.Text(root, width=55, height=12, state='disabled')` — многострочное «окно переписки». `state='disabled'` запрещает пользователю вводить текст в лог.
- `last_message_id = 0` — номер последнего прочитанного сообщения; для long polling.
- `def append_message(message):` — добавляет строку `имя: текст` в лог.
- `chat_log.config(state='normal')` — временно разрешает изменять лог, чтобы добавить сообщение, затем снова блокирует.
- `threading.Thread(target=poll_loop, daemon=True).start()` — запускает фоновый поток, который следит за новыми сообщениями.
- `while joined:` — поток работает, пока идёт вход в чат.
- `def poll_loop():` — функция фонового потока: опрашивает сервер через `/chats/poll`.
- `root.after(0, append_message, message)` — безопасно добавляет сообщение в интерфейс **из фонового потока**. В Tkinter менять окно можно только в главном потоке, поэтому используем `after`.
- `daemon=True` — поток завершится вместе с закрытием окна.
