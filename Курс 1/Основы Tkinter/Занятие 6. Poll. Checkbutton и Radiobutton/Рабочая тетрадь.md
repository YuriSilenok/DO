# Часть 1. Классная работа

### Шаг 1. Вспоминаем, что у нас уже есть

**Что делаем.** Открываем `main.py` из занятия 5. У вас должен остаться этот полный код:

```python
import tkinter as tk
import requests

BASE_URL = 'http://127.0.0.1:8000'
HEALTH_URL = BASE_URL + '/health'


class ChatApp:
    def __init__(self, root):
        self.root = root
        root.title('Анонимный чат')
        root.geometry('420x560')

        self.username = ''
        self.chat_id = 0
        self.password = ''
        self.last_message_id = 0

        self._build_widgets()

    def _build_widgets(self):
        self.welcome = tk.Label(self.root, text='Добро пожаловать в чат!', font=('Arial', 14))
        self.welcome.pack(pady=10)

        self.info = tk.Label(self.root, text='Введите номер чата, имя и пароль')
        self.info.pack()

        self.chat_entry = tk.Entry(self.root, width=20)
        self.chat_entry.pack(pady=2)

        self.name_entry = tk.Entry(self.root, width=20)
        self.name_entry.pack(pady=2)

        self.pass_entry = tk.Entry(self.root, width=20, show='*')
        self.pass_entry.pack(pady=2)

        self.enter_btn = tk.Button(self.root, text='Войти в чат', command=self.on_enter)
        self.enter_btn.pack(pady=10)

        self.check_btn = tk.Button(self.root, text='Проверить сервер', command=self.check_server)
        self.check_btn.pack(pady=5)

        self.chat_view = tk.Text(self.root, height=8, state='disabled')
        self.chat_view.pack(pady=5, padx=5, fill='both', expand=True)

        self.msg_entry = tk.Entry(self.root, width=30)
        self.msg_entry.pack(pady=2)

        self.send_btn = tk.Button(self.root, text='Отправить', command=self.send_message)
        self.send_btn.pack(pady=5)

        self.cmd1 = tk.Button(self.root, text='Привет', command=lambda: self.send_text('Привет!'))
        self.cmd1.pack(pady=1)

        self.cmd2 = tk.Button(self.root, text='Помощь', command=lambda: self.send_auto('Помощь'))
        self.cmd2.pack(pady=1)

    def set_status(self, text):
        self.info.config(text=text)

    def on_enter(self):
        chat_text = self.chat_entry.get()
        name = self.name_entry.get()
        password = self.pass_entry.get()
        if not chat_text or not name or not password:
            self.set_status('Заполните все поля: чат, имя, пароль!')
            return

        try:
            self.set_status('Подключаюсь к чату {0}...'.format(chat_text))
            response = requests.post(
                BASE_URL + '/chats/join',
                json={
                    'chat_id': int(chat_text),
                    'username': name,
                    'password': password,
                },
                timeout=10,
            )
            data = response.json()
            if response.status_code == 200:
                self.username = data['username']
                self.chat_id = data['chat_id']
                self.password = password
                self.set_status(
                    'Вы вошли как {0} в чат №{1}'.format(self.username, self.chat_id)
                )
            else:
                self.set_status('Ошибка {0}: {1}'.format(
                    response.status_code, data.get('detail', 'неизвестная ошибка')
                ))
        except ValueError:
            self.set_status('Номер чата должен быть целым числом!')
        except requests.exceptions.RequestException:
            self.set_status('Ошибка сети: запустите сервер chat_api (python -m chat_api)')

    def check_server(self):
        try:
            response = requests.get(HEALTH_URL, timeout=5)
            if response.status_code == 200:
                self.set_status('Сервер чата работает: {0}'.format(response.json()))
            else:
                self.set_status('Сервер ответил кодом {0}'.format(response.status_code))
        except requests.exceptions.RequestException:
            self.set_status('Ошибка сети: запустите сервер chat_api (python -m chat_api)')

    def append_message(self, text):
        self.chat_view.config(state='normal')
        self.chat_view.insert(tk.END, text + '\n')
        self.chat_view.see(tk.END)
        self.chat_view.config(state='disabled')

    def send_message(self):
        text = self.msg_entry.get().strip()
        if not text:
            self.set_status('Введите текст сообщения')
            return
        if not self.chat_id:
            self.set_status('Сначала войдите в чат!')
            return
        try:
            response = requests.post(
                BASE_URL + '/chats/send',
                json={
                    'chat_id': self.chat_id,
                    'username': self.username,
                    'password': self.password,
                    'text': text,
                },
                timeout=10,
            )
            data = response.json()
            if response.status_code == 200:
                message = data['message']
                self.append_message('{0}: {1}'.format(self.username, message['text']))
                self.msg_entry.delete(0, tk.END)
                self.set_status('Сообщение отправлено (id {0})'.format(message['id']))
            else:
                self.set_status('Ошибка {0}: {1}'.format(
                    response.status_code, data.get('detail', 'неизвестная ошибка')
                ))
        except requests.exceptions.RequestException:
            self.set_status('Ошибка сети: запустите сервер chat_api (python -m chat_api)')

    def send_text(self, text):
        self.msg_entry.delete(0, tk.END)
        self.msg_entry.insert(0, text)
        self.send_message()

    def send_auto(self, text):
        self.set_status('Команда «{0}» будет реализована на занятии 6'.format(text))


app = ChatApp(tk.Tk())
app.root.mainloop()
```

### Шаг 3. Пишем метод опроса сервера poll_once

**Что делаем.** Добавляем в класс `ChatApp` методы:

```python
    def poll_once(self):
        try:
            response = requests.post(
                BASE_URL + '/chats/poll',
                json={
                    'chat_id': self.chat_id,
                    'username': self.username,
                    'password': self.password,
                    'last_message_id': self.last_message_id,
                    'timeout': 2,
                },
                timeout=10,
            )
            data = response.json()
            if response.status_code == 200:
                for m in data['messages']:
                    self.append_message(
                        '{0}: {1}'.format(m['username'], m['text'] or '')
                    )
                    self.last_message_id = m['id']
                return True
        except requests.exceptions.RequestException:
            pass
        return False
```

**Подробнее о запросе `POST /chats/poll`:**
- Зачем нужен: самое главное в `chat_api` — **Long Polling** (долгий опрос). Клиент (наше окно) спрашивает: «дай всё, что новее `last_message_id`». Если сообщений нет сразу — сервер **ждёт** до `timeout` секунд и только потом отвечает. Если за это время кто-то напишет, сервер сразу вернёт новые сообщения.
- Параметры: `chat_id`, `username`, `password` — «ключ» доступа; `last_message_id` — id последнего полученного сообщения (сервер отдаст только более новые); `timeout` — сколько секунд сервер готов ждать (мы ставим 2, сервер разрешает до 60).
- **Почему «long»:** вместо частых пустых запросов клиент делает один «долгий» — это экономно и почти мгновенно реагирует на новые сообщения.

**Что делает `poll_once`:** опрашивает сервер один раз; полученные сообщения добавляет в ленту и запоминает их id в `self.last_message_id`, чтобы не получать их повторно. Возвращает `True/False` — успешно ли прошёл опрос.

**Проверь себя.** Метод добавился в класс (проверим его действие в шаге 5 при запуске).

### Шаг 4. Добавляем фоновый поток и проверку после входа

**Что делаем.** Добавляем в класс метод запуска фонового опроса, а в `on_enter` после успешного входа — вызов запуска:

```python
    def start_polling(self):
        def worker():
            while True:
                if not self.poll_once() and not self.poll_once():
                    time.sleep(1)
                time.sleep(1)
        threading.Thread(target=worker, daemon=True).start()
```

В методе `on_enter`, в блоке `if response.status_code == 200:`, в самый конец (после `self.set_status(...)`) добавляем строчку:

```python
                self.start_polling()
```

**Что здесь происходит:**

- `threading.Thread(target=worker, daemon=True)` — создаёт фоновый поток, который выполняет `worker` параллельно с окном. `daemon=True` значит: поток завершится автоматически вместе с закрытием программы.
- `worker` — бесконечный цикл внутри потока: вызывает `poll_once`, а между вызовами пауза `time.sleep(1)`. Проверка `if not self.poll_once() and not self.poll_once():` — первый вызов «долгий» (сервер ждёт до 2 секунд), второй сразу проверяет, не появились ли сообщения за это время.
- **Запуск из `on_enter`** — опрос начинается только после входа в чат.

**Проверь себя.** После входа в чат сообщения других пользователей появляются в ленте без нажатия кнопок. Для проверки откройте **второе** окно программы (запустите `main.py` ещё раз) и напишите оттуда сообщение — в первом окне оно появится само.

### Шаг 6. Знакомимся с переменными Tk: StringVar и BooleanVar

**Что делаем.** После `root.geometry(...)` (в начале `__init__`) добавляем переменные-настройки:

```python
        self.auto_scroll = tk.BooleanVar(value=True)
        self.notify_mode = tk.StringVar(value='все')
```

**Подробнее о переменных Tkinter:**

**Зачем нужны.** Обычные переменные Python не «видят» изменения из виджетов. Переменные Tk привязаны к виджетам: изменил `.get()` — обновился виджет, изменил виджет — изменилась переменная.

**Типы переменных:**
- `tk.StringVar(value='...')` — строковая переменная (аналог `str`);
- `tk.BooleanVar(value=...)` — логическая (аналог `bool`);
- `tk.IntVar(...)`, `tk.DoubleVar(...)` — для чисел.

**Методы этих переменных:** `.get()` — получить значение, `.set(...)` — установить, `.trace_add(...)` — вызвать функцию при изменении.

**Проверь себя.** Переменные созданы без ошибок.

### Шаг 7. Добавляем Checkbutton и Radiobutton

**Что делаем.** В `_build_widgets`, после ленты сообщений (`self.chat_view...`), добавляем панель настроек:

```python
        self.auto_check = tk.Checkbutton(
            self.root, text='Автопрокрутка ленты',
            variable=self.auto_scroll,
        )
        self.auto_check.pack(pady=2)

        self.radio_frame = tk.Frame(self.root)
        self.radio_frame.pack(pady=2)

        tk.Radiobutton(
            self.radio_frame, text='Все сообщения', value='все',
            variable=self.notify_mode,
        ).pack(side='left', padx=4)

        tk.Radiobutton(
            self.radio_frame, text='Только текст', value='только текст',
            variable=self.notify_mode,
        ).pack(side='left', padx=4)
```

**Подробнее о виджетах:**

**`Checkbutton`** — «флажок»: две позиции (вкл/выкл). Параметр `variable` связывает его с `BooleanVar`. Полезен для настроек вида «автопрокрутка: да/нет».

**`Radiobutton`** — «радиокнопка»: группа, в которой можно выбрать **только один** вариант. `variable` у всех кнопок группы — **одна и та же** `StringVar`; `value` — значение, которое она примет при выборе. Полезен для выбора режима.

**Метод `variable.get()`**: у нас `self.notify_mode.get()` вернёт `'все'` или `'только текст'`, `self.auto_scroll.get()` — `True`/`False`. Используем это на занятии 7, чтобы управлять отображением ленты.

**Проверь себя.** Под лентой появились флажок «Автопрокрутка ленты» и две радиокнопки. Одновременно может быть выбрана только одна из двух.
