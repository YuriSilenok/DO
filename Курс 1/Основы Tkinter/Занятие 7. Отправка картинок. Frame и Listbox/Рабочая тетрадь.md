**Что делаем.** Открываем `main.py` из занятия 6. У вас должен остаться этот полный код (сверяйтесь с прошлой тетрадью):

```python
import tkinter as tk
import requests
import time
import threading

BASE_URL = 'http://127.0.0.1:8000'
HEALTH_URL = BASE_URL + '/health'


class ChatApp:
    def __init__(self, root):
        self.root = root
        root.title('Анонимный чат')
        root.geometry('420x560')

        self.username = ''
        self.chat_id = 0
        self.password = ''
        self.last_message_id = 0
        self.auto_scroll = tk.BooleanVar(value=True)
        self.notify_mode = tk.StringVar(value='все')

        self._build_widgets()

    def _build_widgets(self):
        self.welcome = tk.Label(self.root, text='Добро пожаловать в чат!', font=('Arial', 14))
        self.welcome.pack(pady=10)

        self.info = tk.Label(self.root, text='Введите номер чата, имя и пароль')
        self.info.pack()

        self.chat_entry = tk.Entry(self.root, width=20)
        self.chat_entry.pack(pady=2)

        self.name_entry = tk.Entry(self.root, width=20)
        self.name_entry.pack(pady=2)

        self.pass_entry = tk.Entry(self.root, width=20, show='*')
        self.pass_entry.pack(pady=2)

        self.enter_btn = tk.Button(self.root, text='Войти в чат', command=self.on_enter)
        self.enter_btn.pack(pady=10)

        self.check_btn = tk.Button(self.root, text='Проверить сервер', command=self.check_server)
        self.check_btn.pack(pady=5)

        self.chat_view = tk.Text(self.root, height=8, state='disabled')
        self.chat_view.pack(pady=5, padx=5, fill='both', expand=True)

        self.auto_check = tk.Checkbutton(
            self.root, text='Автопрокрутка ленты',
            variable=self.auto_scroll,
        )
        self.auto_check.pack(pady=2)

        self.radio_frame = tk.Frame(self.root)
        self.radio_frame.pack(pady=2)

        tk.Radiobutton(
            self.radio_frame, text='Все сообщения', value='все',
            variable=self.notify_mode,
        ).pack(side='left', padx=4)

        tk.Radiobutton(
            self.radio_frame, text='Только текст', value='только текст',
            variable=self.notify_mode,
        ).pack(side='left', padx=4)

        self.msg_entry = tk.Entry(self.root, width=30)
        self.msg_entry.pack(pady=2)

        self.send_btn = tk.Button(self.root, text='Отправить', command=self.send_message)
        self.send_btn.pack(pady=5)

        self.cmd1 = tk.Button(self.root, text='Привет', command=lambda: self.send_text('Привет!'))
        self.cmd1.pack(pady=1)

        self.cmd2 = tk.Button(self.root, text='Настройки', command=self.show_settings)
        self.cmd2.pack(pady=1)

    def set_status(self, text):
        self.info.config(text=text)

    def on_enter(self):
        chat_text = self.chat_entry.get()
        name = self.name_entry.get()
        password = self.pass_entry.get()
        if not chat_text or not name or not password:
            self.set_status('Заполните все поля: чат, имя, пароль!')
            return

        try:
            self.set_status('Подключаюсь к чату {0}...'.format(chat_text))
            response = requests.post(
                BASE_URL + '/chats/join',
                json={
                    'chat_id': int(chat_text),
                    'username': name,
                    'password': password,
                },
                timeout=10,
            )
            data = response.json()
            if response.status_code == 200:
                self.username = data['username']
                self.chat_id = data['chat_id']
                self.password = password
                self.set_status(
                    'Вы вошли как {0} в чат №{1}'.format(self.username, self.chat_id)
                )
                self.start_polling()
            else:
                self.set_status('Ошибка {0}: {1}'.format(
                    response.status_code, data.get('detail', 'неизвестная ошибка')
                ))
        except ValueError:
            self.set_status('Номер чата должен быть целым числом!')
        except requests.exceptions.RequestException:
            self.set_status('Ошибка сети: запустите сервер chat_api (python -m chat_api)')

    def check_server(self):
        try:
            response = requests.get(HEALTH_URL, timeout=5)
            if response.status_code == 200:
                self.set_status('Сервер чата работает: {0}'.format(response.json()))
            else:
                self.set_status('Сервер ответил кодом {0}'.format(response.status_code))
        except requests.exceptions.RequestException:
            self.set_status('Ошибка сети: запустите сервер chat_api (python -m chat_api)')

    def append_message(self, text):
        self.chat_view.config(state='normal')
        self.chat_view.insert(tk.END, text + '\n')
        if self.auto_scroll.get():
            self.chat_view.see(tk.END)
        self.chat_view.config(state='disabled')

    def send_message(self):
        text = self.msg_entry.get().strip()
        if not text:
            self.set_status('Введите текст сообщения')
            return
        if not self.chat_id:
            self.set_status('Сначала войдите в чат!')
            return
        try:
            response = requests.post(
                BASE_URL + '/chats/send',
                json={
                    'chat_id': self.chat_id,
                    'username': self.username,
                    'password': self.password,
                    'text': text,
                },
                timeout=10,
            )
            data = response.json()
            if response.status_code == 200:
                message = data['message']
                self.append_message('{0}: {1}'.format(self.username, message['text']))
                self.msg_entry.delete(0, tk.END)
                self.set_status('Сообщение отправлено (id {0})'.format(message['id']))
            else:
                self.set_status('Ошибка {0}: {1}'.format(
                    response.status_code, data.get('detail', 'неизвестная ошибка')
                ))
        except requests.exceptions.RequestException:
            self.set_status('Ошибка сети: запустите сервер chat_api (python -m chat_api)')

    def send_text(self, text):
        self.msg_entry.delete(0, tk.END)
        self.msg_entry.insert(0, text)
        self.send_message()

    def send_auto(self, text):
        self.set_status('Команда «{0}» будет реализована на занятии 6'.format(text))

    def show_settings(self):
        self.set_status(
            'Автопрокрутка: {0}; режим: {1}'.format(
                self.auto_scroll.get(), self.notify_mode.get()
            )
        )

    def poll_once(self):
        try:
            response = requests.post(
                BASE_URL + '/chats/poll',
                json={
                    'chat_id': self.chat_id,
                    'username': self.username,
                    'password': self.password,
                    'last_message_id': self.last_message_id,
                    'timeout': 2,
                },
                timeout=10,
            )
            data = response.json()
            if response.status_code == 200:
                for m in data['messages']:
                    if self.notify_mode.get() == 'только текст' and m['file_url']:
                        self.last_message_id = m['id']
                        continue
                    self.append_message(
                        '{0}: {1}'.format(m['username'], m['text'] or '')
                    )
                    self.last_message_id = m['id']
                return True
        except requests.exceptions.RequestException:
            pass
        return False

    def start_polling(self):
        def worker():
            while True:
                if not self.poll_once() and not self.poll_once():
                    time.sleep(1)
                time.sleep(1)
        threading.Thread(target=worker, daemon=True).start()


app = ChatApp(tk.Tk())
app.root.mainloop()
```

**Класс `ImageTk.PhotoImage(изображение)`:** превращает объект Pillow в картинку Tkinter, которую можно вставить в `Text` через `image_create`.

### Шаг 4. Отправляем картинку на сервер — POST /chats/send_file

**Что делаем.** Добавляем в класс метод `upload_image`:

```python
    def upload_image(self, path):
        if not self.chat_id:
            self.set_status('Сначала войдите в чат!')
            return
        try:
            files = {'file': open(path, 'rb')}
            data = {
                'chat_id': self.chat_id,
                'username': self.username,
                'password': self.password,
            }
            response = requests.post(
                BASE_URL + '/chats/send_file',
                data=data,
                files=files,
                timeout=30,
            )
            response.raise_for_status()
            self.set_status('Картинка отправлена (id {0})'.format(
                response.json()['message']['id']
            ))
        except requests.exceptions.RequestException:
            self.set_status('Ошибка сети или неверный формат картинки')
```

**Подробнее о запросе `POST /chats/send_file`:**
- Зачем нужен: отправляет файл (картинку или видео) в чат как `multipart/form-data`. Именно такой способ умеет читать сервер `chat_api`.
- Параметры `requests.post(url, data=..., files=..., timeout=...)`:
  - `data` — обычные поля формы: `chat_id`, `username`, `password`;
  - `files` — поле с файлом. Формат: `{'file': (имя_файла, файловый_объект, mime-тип)}`. Мы передаём `{'file': open(path, 'rb')}` — requests сам подставит имя и тип;
  - `timeout` — для загрузки файлов нужен больше: `30` секунд.
- **Допустимые типы сервера:** `image/jpeg`, `image/png`, `image/gif`, `video/mp4`, `video/webm`, `video/quicktime`. Если послать не то — сервер вернёт ошибку `415` (в `response.raise_for_status()` превратится в исключение).

**Проверь себя.** С выбранной картинкой `.png`/`.jpg` запрос проходит, в статусе появляется id отправленного файла.

### Шаг 5. Вставляем картинку в ленту (ImageTk.PhotoImage)

**Что делаем.** Добавляем в класс метод, который вставляет уменьшенную картинку в `Text`:

```python
    def add_image_to_view(self, path):
        try:
            image = Image.open(path)
            image.thumbnail((150, 150))
            photo = ImageTk.PhotoImage(image)

            self.chat_view.config(state='normal')
            self.chat_view.image_create(tk.END, image=photo)
            self.chat_view.insert(tk.END, '\n')
            if self.auto_scroll.get():
                self.chat_view.see(tk.END)
            self.chat_view.config(state='disabled')

            if not hasattr(self, '_images'):
                self._images = []
            self._images.append(photo)
        except Exception:
            self.chat_view.config(state='normal')
            self.chat_view.insert(tk.END, '[Не удалось показать картинку]\n')
            self.chat_view.config(state='disabled')
```

**Что здесь происходит:**

- `Image.open(path)` — открывает файл; `thumbnail((150, 150))` уменьшает картинку до 150×150, сохраняя пропорции; `ImageTk.PhotoImage(image)` — превращает её в изображение для Tkinter.
- `self.chat_view.image_create(tk.END, image=photo)` — вставляет картинку в `Text` в конец ленты.
- **Важно:** `Tk` «забывает» картинку, если на неё нет ссылок. Поэтому все созданные `PhotoImage` мы храним в списке `self._images` — иначе картинки исчезнут из окна.
- `try/except` подстраховывает: если файл битый — вместо падения в ленте появится заглушка.

```python
            self.set_status('Ничего не выбрано в списке')
```

**Как проверить правильность выполнения домашней работы**

1. **Выделение.** При выборе строки в списке и нажатии кнопки удаляется именно она.
2. **Пустой выбор.** Если ничего не выбрано — появляется подсказка «Ничего не выбрано в списке».
3. **Картинки по-прежнему работают.** Отправка и отображение картинок работает.

**Если какой-то пункт не выполняется** — проверьте метод `curselection()`: он возвращает кортеж индексов, поэтому берём `selection[0]`.

Если всё выполнено — на следующем занятии соберём всё воедино: добавим окно «О себе» (приведение картинки к круглой аватарке через `Canvas.create_oval` и `GET /files/{id}`), подведём итоги и подготовим проект к защите.
