# Часть 1. Классная работа

### Шаг 1. Вспоминаем, что у нас уже есть

**Что делаем.** Открываем `main.py` из занятия 3. У вас должен остаться этот полный код:

```python
import tkinter as tk
import requests

root = tk.Tk()
root.title('Анонимный чат')
root.geometry('420x560')

USER = ''
CHAT_ID = 0
BASE_URL = 'http://127.0.0.1:8000'

welcome = tk.Label(root, text='Добро пожаловать в чат!', font=('Arial', 14))
welcome.pack(pady=10)

info = tk.Label(root, text='Введите номер чата, имя и пароль')
info.pack()

chat_entry = tk.Entry(root, width=20)
chat_entry.pack(pady=2)

name_entry = tk.Entry(root, width=20)
name_entry.pack(pady=2)

pass_entry = tk.Entry(root, width=20, show='*')
pass_entry.pack(pady=2)

def on_enter():
    try:
        chat_text = chat_entry.get()
        name = name_entry.get()
        password = pass_entry.get()
        info.config(text='Подключаюсь к чату {0}...'.format(chat_text))

        response = requests.post(
            BASE_URL + '/chats/join',
            json={
                'chat_id': int(chat_text),
                'username': name,
                'password': password,
            },
            timeout=10,
        )
        response.raise_for_status()

        data = response.json()
        USER = name
        CHAT_ID = data['chat_id']
        info.config(
            text='Вы вошли как {0} в чат №{1} (пароль запомнен)'.format(
                USER, CHAT_ID
            )
        )
    except ValueError:
        info.config(text='Номер чата должен быть целым числом!')
    except requests.exceptions.RequestException:
        info.config(text='Ошибка сети: запустите сервер chat_api (python -m chat_api)')

enter_btn = tk.Button(root, text='Войти в чат', command=on_enter)
enter_btn.pack(pady=10)

root.mainloop()
```

### Шаг 4. Добавляем метод on_enter в класс и обрабатываем HTTP-ошибки

**Что делаем.** Добавляем внутрь класса `ChatApp` (после метода `set_status`) метод входа в чат:

```python
    def on_enter(self):
        chat_text = self.chat_entry.get()
        name = self.name_entry.get()
        password = self.pass_entry.get()
        if not chat_text or not name or not password:
            self.set_status('Заполните все поля: чат, имя, пароль!')
            return

        try:
            self.set_status('Подключаюсь к чату {0}...'.format(chat_text))
            response = requests.post(
                BASE_URL + '/chats/join',
                json={
                    'chat_id': int(chat_text),
                    'username': name,
                    'password': password,
                },
                timeout=10,
            )
            data = response.json()
            if response.status_code == 200:
                self.username = data['username']
                self.chat_id = data['chat_id']
                self.password = password
                self.set_status(
                    'Вы вошли как {0} в чат №{1}'.format(self.username, self.chat_id)
                )
            else:
                self.set_status('Ошибка {0}: {1}'.format(
                    response.status_code, data.get('detail', 'неизвестная ошибка')
                ))
        except ValueError:
            self.set_status('Номер чата должен быть целым числом!')
        except requests.exceptions.RequestException:
            self.set_status('Ошибка сети: запустите сервер chat_api (python -m chat_api)')
```

**Что здесь нового:**

- `response.status_code` — код ответа HTTP (`200`, `403`, `404`, `422`...). Теперь мы **сами** смотрим на код вместо `raise_for_status()`.
- `data.get('detail', ...)` — в теле ошибки сервер пишет человекочитаемый текст; `.get` вернёт текст или значение по умолчанию.

**Проверь себя.** Вход работает как раньше; при неверном пароле для существующего имени появляется «Ошибка 403: В чате ... уже существует пользователь ...».

### Шаг 5. Проверяем здоровье сервера запросом GET /health

**Что делаем.** Добавляем в класс `ChatApp` метод проверки сервера и кнопку «Проверить сервер»:

```python
    def check_server(self):
        try:
            response = requests.get(HEALTH_URL, timeout=5)
            if response.status_code == 200:
                self.set_status('Сервер чата работает: {0}'.format(response.json()))
            else:
                self.set_status('Сервер ответил кодом {0}'.format(response.status_code))
        except requests.exceptions.RequestException:
            self.set_status('Ошибка сети: запустите сервер chat_api (python -m chat_api)')
```

И в методе `_build_widgets`, после создания кнопки входа, добавляем:

```python
        self.check_btn = tk.Button(self.root, text='Проверить сервер', command=self.check_server)
        self.check_btn.pack(pady=5)
```

**Подробнее о запросе `GET /health`:**
- Зачем нужен: это «проверка работоспособности» сервера `chat_api`. Если сервер жив — он отвечает `{"status": "ok"}`. Отдельный запрос `health` позволяет понять: проблема на стороне сервера или в нашем коде.
- Параметры `requests.get(url, ...)`: `url` — адрес, `timeout` — количество секунд ожидания. Именно `timeout` защищает программу от «вечного ожидания» неработающего сервера.

**Проверь себя.** Кнопка «Проверить сервер» показывает статус сервера; при выключенном сервере — понятное сообщение.
