# Рабочая тетрадь. Занятие 4. Калькулятор на сложение

## Цель занятия
Сделать простую программу для сложения двух чисел.

## Шаг 1. Создаём поля и кнопку
```python
import tkinter as tk

root = tk.Tk()
root.title('Калькулятор')
root.geometry('300x220')

label1 = tk.Label(root, text='Число 1:')
label1.pack()
entry1 = tk.Entry(root)
entry1.pack()

label2 = tk.Label(root, text='Число 2:')
label2.pack()
entry2 = tk.Entry(root)
entry2.pack()

result = tk.Label(root, text='Результат: ')
result.pack()

def add_numbers():
    a = int(entry1.get())
    b = int(entry2.get())
    total = a + b
    result.config(text=f'Результат: {total}')

button = tk.Button(root, text='Сложить', command=add_numbers)
button.pack()

root.mainloop()
```
