# Рабочая тетрадь. Занятие 3. Entry и ввод текста

## Цель занятия
Научиться получать текст от пользователя через поле ввода.

## Шаг 1. Поле ввода
```python
import tkinter as tk

root = tk.Tk()
root.title('Поле ввода')
root.geometry('300x200')

entry = tk.Entry(root)
entry.pack()

root.mainloop()
```

## Шаг 2. Кнопка для чтения текста
```python
import tkinter as tk

root = tk.Tk()
root.title('Приветствие')
root.geometry('300x200')

label = tk.Label(root, text='Введите имя:')
label.pack()

entry = tk.Entry(root)
entry.pack()

button = tk.Button(root, text='Поприветствовать')
button.pack()

root.mainloop()
```

## Добавляем действие по кнопке
```python
import tkinter as tk

root = tk.Tk()
root.title('Приветствие')
root.geometry('300x200')

label = tk.Label(root, text='Введите имя:')
label.pack()

entry = tk.Entry(root)
entry.pack()

result = tk.Label(root, text='')
result.pack()

def say_hello():
    name = entry.get()
    result.config(text=f'Привет, {name}!')

button = tk.Button(root, text='Поприветствовать', command=say_hello)
button.pack()

root.mainloop()
```
